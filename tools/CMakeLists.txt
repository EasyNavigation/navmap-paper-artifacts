cmake_minimum_required(VERSION 3.14)
project(meshstats)
set(CMAKE_CXX_STANDARD 23)
find_package(navmap_ros REQUIRED)
find_package(navmap_core REQUIRED)
find_package(PCL REQUIRED COMPONENTS io common)
find_package(pcl_conversions REQUIRED)
add_executable(meshstats meshstats.cpp)
target_include_directories(meshstats PRIVATE ${PCL_INCLUDE_DIRS} ${pcl_conversions_INCLUDE_DIRS})
target_link_libraries(meshstats navmap_ros::navmap_ros navmap_core::navmap_core ${PCL_LIBRARIES})
add_executable(lidar_sim lidar_sim.cpp)
target_include_directories(lidar_sim PRIVATE ${PCL_INCLUDE_DIRS})
target_link_libraries(lidar_sim navmap_core::navmap_core ${PCL_LIBRARIES})
add_executable(dist_dump dist_dump.cpp)
target_link_libraries(dist_dump navmap_ros::navmap_ros navmap_core::navmap_core)
add_executable(cloud_to_navmap cloud_to_navmap.cpp)
target_include_directories(cloud_to_navmap PRIVATE ${PCL_INCLUDE_DIRS})
target_link_libraries(cloud_to_navmap navmap_ros::navmap_ros navmap_core::navmap_core ${PCL_LIBRARIES})
