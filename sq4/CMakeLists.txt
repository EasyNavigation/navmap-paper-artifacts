cmake_minimum_required(VERSION 3.14)
project(navmap_paper_sq4)
set(CMAKE_CXX_STANDARD 23)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(navmap_core REQUIRED)
find_package(navmap_ros REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io)
add_executable(sq4 sq4.cpp)
target_include_directories(sq4 PRIVATE ${PCL_INCLUDE_DIRS})
target_link_libraries(sq4 navmap_core::navmap_core navmap_ros::navmap_ros ${nav_msgs_TARGETS}
  ${PCL_LIBRARIES})
